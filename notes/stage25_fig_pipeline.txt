class _H:
    R = re.compile(r"^[\d/∕]*((국어|언어|영역|영|역|홀수형|짝수형|\(?화법과작문\)?|\(?언어와매체\)?|\(?독서\)?|\(?문학\)?|\(?[AB]형\)?)[\d/∕]*)*$")
    def match(self, t):
        u = re.sub(r"\s", "", t)
        if self.R.match(u): return True
        if len(u) <= 3 and not re.search(r"[가-힣A-Za-z]", u): return True
        if len(u) <= 22 and re.match(r"^[\d/∕]*(홀수형|짝수형)?[\d/∕]*(국어|언어)?영역", u) and not re.search(r"[.?]$", u): return True
        if re.search(r"문제지[\d]*$", u) and len(u) < 50: return True
        return False
HDR = _H()
def body(L, pg, col):
    hd = lambda l: HDR.match(l["t"]) or (l["top"] < 140 and len(l["t"].strip()) <= 26 and not re.match(r"^\s*(\[\s*\d|\d{1,2}\s*\.)", l["t"]))
    B = [l for l in L if l["pg"]==pg and l["col"]==col and not hd(l)]
    allb = [l for l in L if l["col"]==col and not HDR.match(l["t"]) and "저작권" not in l["t"] and "확인 사항" not in l["t"]]
    bots = sorted(l.get("bot", l["top"]+14) for l in allb)
    xs0 = sorted(l["x0"] for l in allb); xs1 = sorted(l.get("x1", l["x0"]+250) for l in allb)
    H = [l for l in L if l["pg"]==pg and l["col"]==col and hd(l) and l["top"] < 140]

def unrule(im):
    # 가장자리의 세로 단 구분선(높이의 60% 이상 이어진 검은 세로줄)을 지운다
    W, H = im.size; px = im.load(); im = im.copy(); px = im.load()
    for xs in (range(0, min(14, W)), range(max(0, W-14), W)):
        for x in xs:
            dark = sum(1 for y in range(0, H, 2) if px[x, y] < 140)
            if dark > 0.6 * (H/2):
                for y in range(H): px[x, y] = 255
    return im
def trim(im, pad=10, thr=235):
    im = unrule(im)
    inv = ImageOps.invert(im.point(lambda v: 255 if v>thr else v))
    bb = inv.getbbox()
    if not bb: return im
    x0,y0,x1,y1 = bb
    return im.crop((max(0,x0-pad),max(0,y0-pad),min(im.width,x1+pad),min(im.height,y1+pad)))
def seg_img(ex, L, pg, col, top, bot):

    return im.crop((int(x0*SC), y0, int(x1*SC), y1))

    o.save(out, optimize=True); return out, segs, o.size

def cuts(img):
    a = np.asarray(img); white = (a > 225).all(axis=1)
    res = []; y = 0; H = len(white)
    while y < H:
        if white[y]:
            y0 = y
            while y < H and white[y]: y += 1
            if y - y0 >= 3: res.append((y0 + y)//2)
        else: y += 1
    return res
