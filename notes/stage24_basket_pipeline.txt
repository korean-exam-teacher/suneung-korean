/* 담은 문항을 지문별로 묶는다. 묶음 순서는 처음 담은 순서, 묶음 안에서는 원래 문항 번호 순. */
function bkGroups(){
  const groups = [], map = new Map();
  bkValidKeys().forEach(k=>{
    const it = itemByKey(k);
    const solo = !it.지문범위 || it.지문범위 === '(독립 문항)';
    const gk = solo ? 'solo||'+k : `${it.exam_id}||${it.영역}||${it.지문범위}`;
    if(!map.has(gk)){
      const g = {gk, examId:it.exam_id, 영역:it.영역, 지문범위:it.지문범위, solo, 소재:it.소재, 분야:it.분야, 작품:it.작품||[], 명칭:it.명칭, items:[]};
      map.set(gk, g); groups.push(g);
    }
    map.get(gk).items.push(it);
  });
  const num = new Map(); let n = 0;
  groups.forEach(g=>{
    g.items.sort((a,b)=>a.문항번호-b.문항번호);
    g.start = n+1; g.items.forEach(it=>num.set(itemKey(it), ++n)); g.end = n;
  });
  return {groups, num, total:n};
}

  if(/\[(그림|사진|표)[^\]]*\]/.test(t)) return 'fig';
  return '';
}

function bkFontInstalled(name){
  if(bkFontCache.has(name)) return bkFontCache.get(name);
  let ok = false;
  try{
    const c = document.createElement('canvas').getContext('2d');
    const sample = '가나다라 국어 영역 ABCabc 123 ①②';
    ok = ['monospace','sans-serif','serif'].some(base=>{

      c.font = `40px "${name.replace(/"/g,'')}", ${base}`; return Math.abs(c.measureText(sample).width - w0) > 0.5;
    });
  }catch(e){}
  bkFontCache.set(name, ok); return ok;
}

function bkChoiceCols(ch, g, pt){
  const colMm = (g.w - 2*g.mx - g.gap)/2 - 2;
  const cpl = colMm / (pt*0.3528*1.02);          // 한 줄에 들어가는 대략의 글자 수(한글 기준)
  const maxLen = Math.max(...ch.map(c=>[...c].length));
  for(const k of [5, 2]) if(maxLen + 2.2 <= cpl/k) return k;
  return 1;
}

function bkBlocks(groups, num, g, pt){
  const B = [];
  const srcOn = basket.src;
  groups.forEach(gr=>{
    if(!gr.solo){
      const range = bkRangeLabel(gr);

      let t = getPassage(gr.examId, gr.영역, gr.지문범위);

      } else if(!t){

      } else {

        notes.forEach(n=>{ t = t.replace(n[2], ''); });

        if(poemOnly){
          const lines = t.replace(/^\n+|\s+$/g,'').split('\n');
          B.push({t:'text', tok:lines, sep:'\n', make:(i,j,cont,brk)=>{
            let seg = lines.slice(i,j); if(cont) while(seg.length && !seg[0].trim()) seg = seg.slice(1);
            return `<div class="poem">${escapeHtml(seg.join('\n'))}</div>`; }});
        } else {
          t.trim().split(/\n+/).map(x=>x.trim()).filter(Boolean).forEach(x=>{
            const head = /^\((가|나|다|라|마)\)$|^\[[A-Z]\]$|^<[^>]+>$|^\[(앞부분|중략|뒷부분)/.test(x) || x.length < 8;
            const tok = x.split(' ');
            B.push({t:'text', tok, sep:' ', make:(i,j,cont,brk)=>
              `<div class="pp${head?' ph':''}${cont?' cont':''}${brk?' brk':''}">${escapeHtml(tok.slice(i,j).join(' '))}</div>`});
          });
        }
        notes.forEach(n=>B.push({t:'unit', html:`<div class="pnote">${escapeHtml(n[2])}</div>`}));
        if(pst === 'fig') B.push({t:'unit', html:`<div class="pnote qna">※ 지문 속 [그림]은 이 자료에 싣지 못했습니다. 원본 문제지를 확인하세요.</div>`});
      }
    }
    gr.items.forEach(it=>{
      const n = num.get(itemKey(it));
      const src = srcOn ? ` <span class="src">${gr.solo ? `${escapeHtml(it.명칭)} ` : '원래 '}${it.문항번호}번</span>` : '';
      const stem = `<div class="qs"><span class="qn">${n}.</span>${escapeHtml(bkStem(it, null))}${src}</div>`;
      let bogiHtml = '', bogiPart = null;
      if(bkBogiMissing(it)){
        bogiHtml = `<div class="qna">※ 이 문항의 제시 자료(&lt;보기&gt; 등)는 그림·표라 이 자료에 싣지 못했습니다. 원본 문제지를 확인하세요.</div>`;
        bogiPart = {t:'unit', html:`<div class="q part">${bogiHtml}</div>`};
      } else if(it.보기){

        const label = m ? m[1] : '';
        const body = m ? it.보기.slice(m[0].length) : it.보기;
        const lab = label ? `<div class="qbl">&lt; ${escapeHtml(label)} &gt;</div>` : '';
        bogiHtml = `<div class="qb">${lab}<div class="qbt">${escapeHtml(body)}</div></div>`;
        const tok = body.split(' ');
        bogiPart = {t:'text', tok, sep:' ', make:(i,j,cont,brk)=>
          `<div class="qb${cont?' cont':''}${brk?' brk':''}">${cont?'':lab}<div class="qbt">${escapeHtml(tok.slice(i,j).join(' '))}</div></div>`};
      }
      const ch = bkChoices(it);
      let chHtml;
      if(ch){
        const k = bkChoiceCols(ch, g, pt);
        chHtml = `<div class="qc" style="grid-template-columns:repeat(${k},minmax(0,1fr))">${ch.map((c,i)=>`<div>${CIRCLED[i+1]} ${escapeHtml(c)}</div>`).join('')}</div>`;
      } else chHtml = `<div class="qna">※ 이 문항의 선택지는 그림·도식이라 이 자료에 싣지 못했습니다. 원본 문제지를 확인하세요.</div>`;
      const parts = [{t:'unit', html:`<div class="q">${stem}</div>`}];
      if(bogiPart) parts.push(bogiPart);
      parts.push({t:'unit', html:`<div class="q part">${chHtml}</div>`});
      B.push({t:'unit', html:`<div class="q">${stem}${bogiHtml}${chHtml}</div>`, parts});
    });
  });
  return B;
}
function bkLayoutSync(pt){
  const {groups, num, total} = bkGroups();
  const area = document.getElementById('printArea');
  const g = bkGeom(basket.paper);
  area.innerHTML = '';
  area.style.setProperty('--pf', pt+'pt');
  area.style.setProperty('--pfam', bkFaceCss());
  const title = basket.title.trim();
  const sheets = [];
  let cols, cur, forced = 0;
  const newSheet = ()=>{ const sh = bkEl(sheetHtml(sheets.length, g, title, total)); area.appendChild(sh); sheets.push(sh); cols = sh.querySelectorAll('.sh-col'); cur = cols[0]; };
  const nextCol = ()=>{ if(cur === cols[0]) cur = cols[1]; else newSheet(); };
  const used = ()=>{ const l = cur.lastElementChild; if(!l) return 0; return l.getBoundingClientRect().bottom - cur.getBoundingClientRect().top; };
  const cap = ()=>cur.getBoundingClientRect().height - 1;
  const over = ()=>used() > cap();
  const lineH = pt*1.62*96/72;
  const tryPut = node=>{ cur.appendChild(node); if(!over()) return true; cur.removeChild(node); return false; };
  newSheet();
  function placeText(b){
    let i = 0, cont = false; const n = b.tok.length;
    while(i < n){
      if(tryPut(bkEl(b.make(i, n, cont, false)))) return;
      let lo = i+1, hi = n-1, best = -1;
      while(lo <= hi){
        const mid = (lo+hi)>>1; const p = bkEl(b.make(i, mid, cont, true));
        cur.appendChild(p); const ok = !over(); cur.removeChild(p);
        if(ok){ best = mid; lo = mid+1; } else hi = mid-1;
      }
      if(best > i){ cur.appendChild(bkEl(b.make(i, best, cont, true))); i = best; cont = true; nextCol(); }
      else if(!cur.lastElementChild){ cur.appendChild(bkEl(b.make(i, i+1, cont, true))); i++; cont = true; forced++; nextCol(); }
      else nextCol();
    }
  }
  function place(b){
    if(b.t === 'text'){ placeText(b); return; }
    if(b.t === 'head'){
      const node = bkEl(b.html); cur.appendChild(node);
      if(over() || used() + lineH*3 > cap()){ cur.removeChild(node); if(cur.lastElementChild) nextCol(); cur.appendChild(bkEl(b.html)); }
      return;
    }
    if(tryPut(bkEl(b.html))) return;
    if(cur.lastElementChild){ nextCol(); if(tryPut(bkEl(b.html))) return; }
    if(b.parts){ b.parts.forEach(place); return; }
    cur.appendChild(bkEl(b.html)); forced++; nextCol();
  }
  bkBlocks(groups, num, g, pt).forEach(place);
  while(sheets.length > 1 && !sheets[sheets.length-1].querySelector('.sh-col > *')){ sheets.pop().remove(); }
  const pages = sheets.length;
  if(basket.ansKey) bkAnswerSheets(area, g, groups, num, pages);
  return {pages, pt, paper:basket.paper, forced, total, keyPages: area.children.length - pages};
}

async function bkLayoutAt(pt){
  let r = bkLayoutSync(pt);
  for(let i = 0; i < 3 && document.fonts && document.fonts.status === 'loading'; i++){
    await document.fonts.ready; r = bkLayoutSync(pt);
  }
  return r;
}
async function bkComputeLayout(){
  const sig = bkSig();
  if(bkLayout && bkLayoutSig === sig) return bkLayout;
  if(bkLayoutBusy && bkLayoutBusy.sig === sig) return bkLayoutBusy.p;
  const run = ++bkLayoutRun;
  const p = (async()=>{
    if(document.fonts) await document.fonts.ready;
    const steps = basket.font === 'auto' ? BK_FONT_STEPS : [Number(basket.font)];
    let r = null;
    for(const pt of steps){
      r = await bkLayoutAt(pt);
      if(run !== bkLayoutRun) return null;
      if(r.pages <= BK_MAX_PAGES) break;
    }
    bkLayout = r; bkLayoutSig = sig;
    return r;
  })();
  bkLayoutBusy = {sig, p};
  try{ return await p; } finally { if(bkLayoutBusy && bkLayoutBusy.p === p) bkLayoutBusy = null; }
}
