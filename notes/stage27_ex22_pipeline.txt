def join(ls, verse=False):
    out = ''
    for i, l in enumerate(ls):
        t = l['t']
        if i == 0: out = t; continue
        p = ls[i-1]; k = (p['pg'], p['col'])

        short = p['x1'] < colR[k] - 14

        if gap > 30: out += '\n\n' + t if verse else '\n' + t
        elif short or (indent and not verse and re.match(r'^\S', t) and p['t'].rstrip().endswith(('.', '다', '?', '!', '”', '’', ')'))):
            out += '\n' + t
        else:
            out += (' ' if p.get('te') else '') + t
    return out
